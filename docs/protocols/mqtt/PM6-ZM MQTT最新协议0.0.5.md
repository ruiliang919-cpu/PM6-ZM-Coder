  // 从 0 开始，长度 845 的 bool 数组
  // 根据遥信协议中 0x0000 ~ 0X0B47，一共 845 个点位
  // true 生效 / false 不生效
  // 例如：前三个点位是 true, true, false，表示出现PM6K通讯故障-1#和PM6K通讯故障-2#，但是不出现PM6K通讯故障-3#
    "data": [true, true, false, ....]
